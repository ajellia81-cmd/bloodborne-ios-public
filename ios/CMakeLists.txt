cmake_minimum_required(VERSION 3.24)
project(BloodborneIOSCPU LANGUAGES C CXX ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  message(FATAL_ERROR "This target must use the iOS SDK")
endif()
set(CMAKE_C_STANDARD 11)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)
add_subdirectory(${SDL_SOURCE} sdl EXCLUDE_FROM_ALL)
add_subdirectory(${ZYDIS_SOURCE} zydis EXCLUDE_FROM_ALL)
file(GLOB CPU_SOURCES CONFIGURE_DEPENDS ${RUNTIME_SOURCE}/recomp/src/cpu/*.c ${RUNTIME_SOURCE}/recomp/src/cpu/*.S)
add_library(bbcpu STATIC ${CPU_SOURCES})
target_include_directories(bbcpu PRIVATE ${RUNTIME_SOURCE}/recomp/src)
target_link_libraries(bbcpu PUBLIC Zydis)
target_compile_options(bbcpu PRIVATE $<$<COMPILE_LANGUAGE:C>:-std=gnu11>)
file(GLOB ATRAC_SOURCES CONFIGURE_DEPENDS ${RUNTIME_SOURCE}/third_party/LibAtrac9/C/src/*.c)
add_library(atrac9 STATIC ${ATRAC_SOURCES})
file(GLOB RUNTIME_SOURCES CONFIGURE_DEPENDS ${RUNTIME_SOURCE}/src/runtime*.c)
add_library(bb-runtime STATIC ${RUNTIME_SOURCES} ${RUNTIME_SOURCE}/src/platform.c ${RUNTIME_SOURCE}/ios/bbgpu_stub.c)
target_include_directories(bb-runtime PUBLIC ${RUNTIME_SOURCE} ${RUNTIME_SOURCE}/src ${RUNTIME_SOURCE}/recomp/src)
target_link_libraries(bb-runtime PUBLIC bbcpu atrac9 SDL3::SDL3-static "-framework Foundation")
add_executable(bb-probe ${RUNTIME_SOURCE}/src/probe.c ${RUNTIME_SOURCE}/src/vulkan_smoke.c)
target_link_libraries(bb-probe PRIVATE bb-runtime)
